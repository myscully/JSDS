          <Tile icon="E" title="강화" desc="의심 행위까지 차단" selected={level === "hardened"} onSelectedChange={() => setLevel("hardened")} />

          <Button variant="secondary">이전</Button>
          <Button>다음</Button>
        </CardFoot>
      </Card>
    </Onboard>
  );
}
