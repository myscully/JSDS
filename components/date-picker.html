React · @jiran/ds-react
import { useState } from "react";
import { Button, Calendar, type DateRange } from "@jiran/ds-react";

const TODAY = new Date(2026, 8, 7);
const daysAgo = (n: number) => new Date(TODAY.getFullYear(), TODAY.getMonth(), TODAY.getDate() - n);

export default function Example() {
  const [month, setMonth] = useState(new Date(2026, 8, 1));
  const [range, setRange] = useState<DateRange>({ start: new Date(2026, 8, 1), end: TODAY });
  const pick = (d: Date) => setRange((r) => (!r.end ? (d < r.start ? { start: d, end: r.start } : { start: r.start, end: d }) : { start: d, end: null }));
  return (
    <Calendar
      month={month}
      onMonthChange={setMonth}
      mode="range"
      value={range}
      onSelect={pick}
      today={TODAY}
      footer={
        <>
          <Button size="sm" variant="text" onClick={() => setRange({ start: daysAgo(6), end: TODAY })}>최근 7일</Button>
          <Button size="sm" variant="text" onClick={() => setRange({ start: daysAgo(29), end: TODAY })}>최근 30일</Button>
          <Button size="sm">적용</Button>
        </>
      }
    />
  );
}

Props

HTML 에서는 클래스와 속성으로, React(@jiran/ds-react)에서는 같은 이름의 prop(variant · size …)으로 변형·상태를 지정합니다.

Prop / ClassType · ValuesDefaultDescription
.datepickercontainer—label + .field-btn(+ .calendar). 열림은 .open
.field-btnbutton—값 표시 트리거. 기간은 .sep 로 구분
.calendarpopover—.cal-head + .cal-grid(.dow ×7, .day ×42) + .cal-foot
.day.on / .in-range / .today / .mutedstate—선택 / 기간 사이 / 오늘 / 다른 달
disabledattribute—선택 불가 날짜

Usage

React: @jiran/ds-react 패키지를 설치하고 style.css 를 한 번 import 하면 위 React 탭 코드를 그대로 쓸 수 있습니다(React Package). HTML: assets/style.css 를 연결하고(일부만 필요하면 아래 컴포넌트 CSS 복사), 동작이 필요하면 assets/ds.js 를 함께 넣습니다 — 이 페이지의 프리뷰가 그 스크립트로 동작합니다. 색은 전부 토큰(var(--accent) 등)을 참조하므로 --accent-* 만 제품 컬러로 바꾸면 됩니다.
