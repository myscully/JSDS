Design TokenN

이 가이드가 쓰는 토큰 원본(CSS 변수 · Light/Dark)과 Radius · Motion · Accessibility 규칙입니다. --accent-* 만 제품 컬러로 바꾸면 나머지는 그대로 씁니다. 색의 의미와 사용 규칙은 Foundations › Colors 를 참고하세요.

Token Layers

Primitive
--cool-neutral-* --blue-* --red-* --orange-*… 참조 시스템의 Atomic 램프 13개를 값과 번호까지 그대로. 숫자가 클수록 밝습니다. 컴포넌트에서 직접 쓰지 않음 (Colors › Atomic)
Semantic
--{그룹}-{하위}-{이름} — 그룹은 foreground · background · surface · line · effect · static 여섯입니다(--foreground-neutral-primary · --surface-brand-heavy · --line-negative-focus …). 79개이고 Figma 변수 경로와 1:1 입니다. 컴포넌트는 이것만 참조하고, 프리미티브 직접 참조는 빌드가 막습니다 (Colors › Semantic)
제품 적용
제품 메인 컬러 HEX 하나로 --prod-40~60 다섯 단계를 생성해 주입 → 나머지 토큰은 공통. 면 위 글자가 늘 흰색이라 제품 색 위에서 흰 글자가 4.5 를 넘어야 합니다

  • 텍스트 대비 4.5:1, 대형 텍스트·UI 요소 3:1. 글자색은 --foreground-* 토큰으로 칠하고, 채운 면 위 글자만 --brand-label(Brand 면) · --static-white(Count) 를 씁니다 — 대비는 빌드가 잽니다

Accent 스케일 생성 (JS)

제품 메인 컬러 HEX 하나로 --prod-40~60 다섯 단계를 만들어 :root 에 주입합니다. 메인 컬러가 상태색과 색상환 15도 미만이면 스케일 대신 무채색 램프를 넣고 data-accent-neutral 을 켭니다. 제품 색에는 제약이 있습니다 — Brand 면 위 글자는 늘 흰색이므로 제품 색 위에서 흰 글자가 4.5 를 넘어야 하고, 빌드가 검사합니다.

function hexToHsl(hex){const n=parseInt(hex.slice(1),16);const r=((n>>16)&255)/255,g=((n>>8)&255)/255,b=(n&255)/255;const mx=Math.max(r,g,b),mn=Math.min(r,g,b);let h=0,s=0;const l=(mx+mn)/2;if(mx!==mn){const d=mx-mn;s=l>.5?d/(2-mx-mn):d/(mx+mn);switch(mx){case r:h=(g-b)/d+(g<b?6:0);break;case g:h=(b-r)/d+2;break;default:h=(r-g)/d+4;}h/=6;}return [h*360,s*100,l*100];}
function hslToHex(h,s,l){s/=100;l/=100;const k=n=>(n+h/30)%12;const a=s*Math.min(l,1-l);const f=n=>l-a*Math.max(-1,Math.min(k(n)-3,Math.min(9-k(n),1)));return "#"+[f(0),f(8),f(4)].map(x=>Math.round(x*255).toString(16).padStart(2,"0")).join("").toUpperCase();}
export function accentScale(hex){const [h,s,l6]=hexToHsl(hex);const up=t=>l6+(88-l6)*t,dn=t=>l6*t;const L={50:96,100:90,200:80,300:up(.55),400:up(.32),500:up(.13),700:dn(.78),800:dn(.58),900:dn(.38)};const S={50:.7,100:.7,200:.7,300:.8,400:.85,500:.9};const out={};for(const [k,l] of Object.entries(L)) out[k]=hslToHex(h,Math.min(100,s*(S[k]||1)),Math.max(0,Math.min(100,l)));out[600]=hex.toUpperCase();return out;}
const NEUTRAL_RAMP={50:"#F7F7F8",100:"#EAEBEC",200:"#C2C4C8",300:"#AEB0B6",400:"#989BA2",500:"#70737C",600:"#5A5C63",700:"#37383C",800:"#2E2F33",900:"#1B1C1E"};
const STATUS_HUES=[0,38,160,199]; // danger · warning · success · info
function clashesWithStatus(hex){const [h,s]=hexToHsl(hex);if(s<20) return false;return STATUS_HUES.some(t=>{const d=Math.abs(h-t)%360;return Math.min(d,360-d)<15;});}
function relLum(hex){const c=[0,1,2].map(i=>parseInt(hex.slice(1+i*2,3+i*2),16)/255).map(v=>v<=.03928?v/12.92:Math.pow((v+.055)/1.055,2.4));return .2126*c[0]+.7152*c[1]+.0722*c[2];}
function ratio(a,b){const [x,y]=[relLum(a)+.05,relLum(b)+.05];return Math.max(x,y)/Math.min(x,y);}
// 메인 컬러 위에 흰 글자가 뜨는지 — 어두운 글자(#171719 = --foreground-neutral-primary)가 더 대비가 높으면 밝은 제품이다
export function isLightAccent(hex){return ratio(hex,"#FFFFFF")<ratio(hex,"#171719");}
export function applyAccent(hex){const neutral=clashesWithStatus(hex);const root=document.documentElement;
  /* --accent-* 가 아니라 --prod-* 로 주입한다 — --prod-* 는 제품이 넣는 입력, --accent-* 는 테마가 해석한 결과 */
  const sc=neutral?NEUTRAL_RAMP:accentScale(hex);
  for(const [k,v] of Object.entries(sc)) root.style.setProperty("--prod-"+k,v);
  if(neutral) root.dataset.accentNeutral=""; else delete root.dataset.accentNeutral;
  /* 밝은 제품은 글자를 어둡게 쓰고 상태가 밝은 쪽으로 간다. 무채색 램프는 늘 흰 글자라 함께 켜지지 않는다 */
  if(!neutral && isLightAccent(sc[600])) root.dataset.accentLight=""; else delete root.dataset.accentLight;}
